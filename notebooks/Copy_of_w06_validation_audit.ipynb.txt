        "# Fallback dataset setup\n",
        "if 'df' in locals() and isinstance(df, pd.DataFrame) and 'ctr' in df.columns:\n",

        "        'impressions_count': np.random.randint(100, 10000, size=n),\n",
        "        'ctr': np.random.uniform(0.01, 0.20, size=n),\n",
        "        'historical_ctr_avg': np.random.uniform(0.01, 0.18, size=n)\n",
        "    })\n",
        "\n",
        "features = ['age_days', 'impressions_count', 'historical_ctr_avg']\n",
        "target = 'ctr'\n",
        "\n",
        "# 1. Naive Random Split\n",
        "np.random.seed(42)\n",
        "shuffled_idx = np.random.permutation(len(df_audit))\n",
        "split_n = int(0.8 * len(df_audit))\n",
        "\n",
        "X_train_rand = df_audit.iloc[shuffled_idx[:split_n]][features]\n",
        "y_train_rand = df_audit.iloc[shuffled_idx[:split_n]][target]\n",
        "X_val_rand = df_audit.iloc[shuffled_idx[split_n:]][features]\n",
        "y_val_rand = df_audit.iloc[shuffled_idx[split_n:]][target]\n",
        "\n",
        "model_rand = XGBRegressor(n_estimators=50, random_state=42)\n",
        "model_rand.fit(X_train_rand, y_train_rand)\n",
        "rmse_rand = np.sqrt(mean_squared_error(y_val_rand, model_rand.predict(X_val_rand)))\n",
        "\n",
        "# 2. Honest Time-Aware Split\n",

        "model_time = XGBRegressor(n_estimators=50, random_state=42)\n",
